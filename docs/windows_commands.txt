# show the tables:
mysql.exe -u root -e "SHOW TABLES FROM iot_metering;"

# describe the table
mysql.exe -u root -e "DESCRIBE iot_metering.meter_aggregates;"

# enter a meter:
mysql -u root -p -e "INSERT INTO iot_metering.meters (meter_id, name, description, created_at_utc) VALUES ('mock-meter-001', 'Mock meter 1', 'Development test meter', UTC_TIMESTAMP());"

# you get an error ERROR 1452 foreign key constraint fails

# create a user and define which user rights it has:
mysql -u root -p -e "CREATE USER IF NOT EXISTS 'iot_meter_backend'@'localhost' IDENTIFIED BY 'test99'; GRANT SELECT, INSERT ON iot_metering.* TO 'iot_meter_backend'@'localhost';"

# check now if the data point exist
mysql.exe -u iot_meter_backend -p -e "SELECT * FROM iot_metering.meters;"

# One-time: token from .env (currently API_TOKEN_INGEST is what the GET endpoint checks)
$TOKEN = "test123"
$B = "http://127.0.0.1:8000/v1/get-meter-data.php"

# Happy path — a day that has readings
curl.exe -i "$B`?date=2026-08-06&meter_id=mock-meter-001" -H "Authorization: Bearer $TOKEN"

# Edge cases — each should give a DIFFERENT status
curl.exe -i "$B`?date=2001-01-01&meter_id=mock-meter-001" -H "Authorization: Bearer $TOKEN"   # 200, slots: []
curl.exe -i "$B`?meter_id=ghost-meter"                    -H "Authorization: Bearer $TOKEN"   # 404
curl.exe -i "$B`?date=2026-08-06"                         -H "Authorization: Bearer $TOKEN"   # 400 meter_id required
curl.exe -i "$B`?date=2026-02-31&meter_id=mock-meter-001" -H "Authorization: Bearer $TOKEN"   # 400 bad date
curl.exe -i "$B`?meter_id=mock-meter-001"                 -H "Authorization: Bearer wrong"    # 401
curl.exe -i "$B`?meter_id=mock-meter-001"                                                     # 401 (no header)
curl.exe -i -X POST "$B`?meter_id=mock-meter-001"         -H "Authorization: Bearer $TOKEN"   # 405 + Allow: GET

# Rate limit: 61 quick requests -> the last ones become 429
1..61 | ForEach-Object { curl.exe -s -o NUL -w "%{http_code} " "$B`?meter_id=mock-meter-001" -H "Authorization: Bearer $TOKEN" }
